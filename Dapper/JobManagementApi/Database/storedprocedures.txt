CREATE PROCEDURE InsertJob(
    @p_CompanyId INT,
    @p_RecruiterId INT,
    @p_Title VARCHAR(500),
    @p_Description TEXT,
    @p_Location text,
    @p_SalaryMin DECIMAL(15,2),
    @p_SalaryMax DECIMAL(15,2),
    @p_JobType VARCHAR(100),
    @p_Experience INT
)
as
BEGIN
    INSERT INTO dbo.Job (
        CompanyId,
        RecruiterId,
        Title,
        Description,
        Location,
        SalaryMin,
        SalaryMax,
        JobType,
        Experience
    )
    VALUES (
        @p_CompanyId,
        @p_RecruiterId,
        @p_Title,
        @p_Description,
        @p_Location,
        @p_SalaryMin,
        @p_SalaryMax,
        @p_JobType,
        @p_Experience
    );
END;

go

CREATE PROCEDURE GetJobSkills(
    @p_JobId INT
)
as
begin
    select s.Id,s.Name
    from dbo.JobSkills js
    inner join dbo.Skills s
	on js.SkillId=s.Id
    where js.JobId=@p_JobId;
END;
