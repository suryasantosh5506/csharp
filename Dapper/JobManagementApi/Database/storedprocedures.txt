CREATE PROCEDURE InsertJob(
    IN p_CompanyId INT,
    IN p_RecruiterId INT,
    IN p_Title VARCHAR(300),
    IN p_Description TEXT,
    IN p_Location VARCHAR(100),
    IN p_SalaryMin DECIMAL(10,2),
    IN p_SalaryMax DECIMAL(15,2),
    IN p_JobType VARCHAR(100),
    IN p_Experience INT
)
BEGIN
    INSERT INTO Job (
        CompanyId,
        RecruiterId,
        Title,
        Description,
        Location,
        SalaryMin,
        SalaryMax,
        JobType,
        Experience
    )
    VALUES (
        p_CompanyId,
        p_RecruiterId,
        p_Title,
        p_Description,
        p_Location,
        p_SalaryMin,
        p_SalaryMax,
        p_JobType,
        p_Experience
    );
END ;

DELIMITER ;

							 in p_SkillId int
							)

	delete from jobskills where jobid=p_JobId and skillid=p_SkillId;
end;
//

DELIMITER //

CREATE PROCEDURE GetJobSkills(
    IN p_JobId INT
)
BEGIN
    select s.Id,s.Name
    from JobSkills js
    inner join Skills s
	on js.SkillId=s.Id
    where js.JobId=p_JobId;
END //

DELIMITER ;
