create table Users(
	Id int primary key identity(1,1),
	Name varchar(100) not null,
	Email varchar(100) not null unique,
	PasswordHash varchar(500) not null,
	Role varchar(20) default 'Candidate' check(Role in ('Candidate','Recruiter','Admin')),
	CreatedAt DateTime2 default getdate(),
);

create table Company(
	Id int identity(1,1) primary key,
	UserId int not null,
	Name varchar(100) not null unique,
	description text,
	location varchar(500) not null,
	website varchar(500) not null,
	createdat datetime2 default getDate(),

	constraint FK_Company_Users
	foreign key (userid) 
	references users(id)
	on delete cascade
	on update cascade
);

create table job(
	id int primary key identity(1,1),
	companyId int not null,
	recruiterId int not null,
	Title varchar(500) not null,
	Description text,
	Location text not null,
	SalaryMin decimal(15,2) not null check(salarymin>0),
	SalaryMax decimal(15,2) not null check(salarymax>0),
	JobType varchar(100) check(JobType in ('FullTime','PartTime','Internship')),
	Experience int not null check(Experience>=0),
	CreatedAt datetime2 default getdate(),

	constraint CK_Job_Salary
	check(SalaryMax>=SalaryMin),

	constraint FK_Job_Company
	foreign key (companyid)
	references company(id)
	on delete cascade
	on update cascade,

	constraint FK_Job_User
	foreign key (recruiterid)
	references users(id)
	on delete no action
	on update no action,
);

create table Application(
	id int identity(1,1) primary key,
	jobid int not null,
	candidateid int not null,
	resumeurl text not null,
	status varchar(100) check(status in ('Applied','Shortlisted','Rejected','Hired')),
	createdat datetime2 default getdate(),

	constraint FK_Application_Job
	foreign key (Jobid)
	references job(id)
	on delete cascade
	on update cascade,

	constraint FK_Application_User
	foreign key (candidateid)
	references users(id)
	on delete no action
    on update no action,

	constraint UQ_Application_Job_Candidate
	unique (jobid, candidateid)
);

create table skills(
	id int primary key identity(1,1),
	name varchar(100) not null unique,
);

create table jobskills(
	jobid int not null,
	skillid int not null,

	primary key(jobid,skillid),

	constraint FK_Jobskills_Job
	foreign key (jobid)
	references job(id)
	on delete cascade
	on update cascade,

	constraint FK_Jobskills_Skill
	foreign key (skillid)
	references skills(id)
	on delete cascade
	on update cascade,
);

create table RecruiterApplications(
    Id int identity(1,1) primary key,
    CandidateId int not null,
    Status varchar(100) not null default 'Pending',
    AppliedAt datetime2 not null default getdate(),
	reason varchar(500) not null,
    ReviewedAt datetime2 null,
    ReviewedBy int null,

    constraint FK_RecruiterApplications_Candidate
        foreign key (CandidateId)
        references users(Id)
        on delete cascade,

    constraint FK_RecruiterApplications_Reviewer
        foreign key(ReviewedBy)
        references users(Id)
        on delete no action,
);
